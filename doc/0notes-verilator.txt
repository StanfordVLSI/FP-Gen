# function "dover" creates a log file e.g. "steveri/ver.log$i"
# Should see result "PASSED" and a variety of different ops/types/operands, not just mostly zeroes!
function dover { 
  ( i=$1; log=steveri/ver.log$i; /bin/rm -rf obj_dir; export SIMULATOR=VERILATOR
    make comp; make run | tee $log   # Do comp separately so as not to pollute the log
    mv -f comp_bb.log steveri/comp_bb.log$i
    mv -f  run_bb.log steveri/run_bb.log$i
    echo $log
  ) |& less -r
}

# Can do this to compare to previous logs i.e. compare log1 vs. log0
i=1; (dover $i | tail -10; diff steveri/ver.log{$((i-1)),$i}) |& less -r

# Do this to review type distribution among files e.g. steveri/ver.log[4567]
# There should be twice as many "Random" types as any other given type
function types { 
   sed 's/= */=/g' $1 | xargs -n 1 | awk -F= '/^Type/{print $2}' | sort | uniq -c | sort -nr
}
i=7; types steveri/vcs.log$i
for f in steveri/ver.log?; do (echo $f; types $f ; echo ""); done | less

# Comparable vcs test generates e.g. "steveri/vcs.log6" for comparison with e.g. steveri/ver.log6"
function dovcs { 
  ( i=$1; log=steveri/vcs.log$i; /bin/rm -rf csrc simv simv.daidr; export SIMULATOR=VCS
    make comp; make run | tee $log  # Do comp separately so as not to pollute the log
    mv -f comp_bb.log steveri/comp_bb.log$i
    mv -f  run_bb.log steveri/run_bb.log$i
    echo $log
  ) |& less -r
}

# TYPE DISTRIBUTION TEST should be 2x weighted towards "Random" type
function types { awk -F= '/Type.=/{print $NF}' $1 | sort | uniq -c | sort -nr; }
i=7; types steveri/ver.log$i
i=7; types steveri/vcs.log$i
for f in steveri/vcs.log[3456]; do (echo $f; types $f ; echo ""); done
for f in steveri/ver.log[3456]; do (echo $f; types $f ; echo ""); done

# OPCODE DISTRIBUTION TEST (00,01,10,11) = (NOP,MUL,ADD,MAD)
function opcodes { awk '/opcode:/{print $NF}' $1 | sort | uniq -c | sort -nr; }
opcodes steveri/ver.log$i
for f in steveri/ver.log?; do (echo $f; opcodes $f ; echo ""); done
for f in steveri/vcs.log?; do (echo $f; opcodes $f ; echo ""); done

# HOTBIT DISTRIBUTION should be heavily weighted towards hotbit=0
for f in steveri/vcs.log?; do (echo $f; grep hotbit= $f | xargs -n 1 | grep hotbit | sort | uniq -c; echo ""); done
